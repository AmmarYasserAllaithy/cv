% -- Headers

\newcommand{\quadHeader}[4]{
    \begin{tabular*}{1.0\textwidth}[t]{p{.8\textwidth}@{\extracolsep{\fill}}r}
        \textbf{\textls[64]{#1}} & {\small#2}\vspace{2pt} \\
        \textit{\small\textls[40]{#3}} & \textit{\small#4} \\
    \end{tabular*}\vspace{-4pt}
}

\newcommand{\tripleHeader}[3]{%
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}%
        \textbf{\textls[50]{#1}} | \textit{\color{dimmed}\small#2} & {\small#3}%
    \end{tabular*}\vspace{2pt}%
}
